# Development & test dependencies — NOT installed into production images.
# **Intent, not fact**: CI installs `requirements-dev.lock`.
#
# Install locally with:  pip install -r requirements-dev.lock
# (or -r requirements.txt -r requirements-dev.txt to resolve fresh, which
# is how you find out that a floor needs raising)
-r requirements.txt

pytest>=8.0.0
# Upper bound is deliberate. Whether a locally defined `event_loop` fixture
# overrides the plugin's own changed at the 1.0 boundary: below it the override
# is honoured, from 1.0 it was removed from the API and a redefinition does
# nothing at all. Unpinned, the suite's behaviour depended on whatever pip last
# resolved, which is not a thing test results should depend on.
pytest-asyncio>=1.0,<2.0
pytest-cov>=4.1.0

# Alembic owns the Postgres schema, and is deliberately NOT in the runtime
# image: migrations are applied by a person or a deploy step, never by the
# application on boot. Charter rule 11 — the app declares the revision it
# needs and fails closed; it does not heal itself. Brings SQLAlchemy, which
# is why the runtime does not need it.
alembic==1.18.5
# SQLAlchemy's async engine runs the sync DBAPI shim on a greenlet, and the
# dependency is an extra rather than a default — without it `alembic upgrade
# head` dies with "the greenlet library is required". Named through the extra
# so the reason travels with it. The alternative was psycopg for migrations
# and asyncpg for the app: two drivers' behaviour to know, for one database.
sqlalchemy[asyncio]>=2.0

# The compose file and the deploy workflow are read as data by
# tests/unit/test_migrate_service.py. It arrives transitively today, which is
# not a reason to leave it undeclared: a test that skips when its dependency
# is missing is a test nobody reads.
PyYAML>=6.0
